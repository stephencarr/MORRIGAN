cmake_minimum_required(VERSION 3.25)
project(morrigan_lira C)
if(ANDROID AND ANDROID_NATIVE_API_LEVEL LESS 28)
    message(FATAL_ERROR "This pinned Pd uses glob/globfree; Android API 28 or newer is required")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(MOBILE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(LIBPD_SOURCE_DIR "${MOBILE_ROOT}/.cache/libpd" CACHE PATH "Pinned libpd checkout")
if(NOT EXISTS "${LIBPD_SOURCE_DIR}/pure-data/src/m_pd.h")
    message(FATAL_ERROR "Run python3 mobile/tools/prepare_lira.py before configuring native code")
endif()
set(PD_UTILS OFF CACHE BOOL "" FORCE)
set(PD_EXTRA OFF CACHE BOOL "" FORCE)
set(PD_MULTI OFF CACHE BOOL "" FORCE)
set(PD_LOCALE OFF CACHE BOOL "" FORCE)
set(LIBPD_STATIC ON CACHE BOOL "" FORCE)
set(LIBPD_SHARED OFF CACHE BOOL "" FORCE)
find_package(Threads REQUIRED)
add_subdirectory("${LIBPD_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/libpd")
add_library(morrigan_core STATIC morrigan_engine.c)
target_include_directories(morrigan_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(morrigan_core PUBLIC libpd_static Threads::Threads m ${CMAKE_DL_LIBS})
# The finite-value safety checks must not be optimized away by fast-math.
target_compile_options(morrigan_core PRIVATE -fno-fast-math -Wall -Wextra)
if(ANDROID)
    add_library(morrigan_lira SHARED morrigan_jni.c)
    target_link_libraries(morrigan_lira PRIVATE morrigan_core)
    target_link_options(morrigan_lira PRIVATE "-Wl,-z,max-page-size=16384")
else()
    enable_testing()
    add_executable(morrigan_native_test "${MOBILE_ROOT}/tests/native/render_test.c")
    target_link_libraries(morrigan_native_test PRIVATE morrigan_core)
    target_compile_options(morrigan_native_test PRIVATE -fno-fast-math -Wall -Wextra)
endif()
